## 一、app.js 文件中心改造

### 已实现 ✅

1. **`DEFAULT_FACTORY_ID` fallback**: `"factory-001"` → `"user"`
2. **`KNOWN_NIFI_DIRS` 硬编码数组** → `KNOWN_NIFI_SUBDIRS` 相对子目录名
3. **`state.rootDir` 动态推导**: 从 API 返回的 `storagePath` 中自动找 `nifi-data` 或 `real_nifi_data`
4. **根节点显示名**: 适配 `real_nifi_data` 场景

### 工作原理

```javascript
function buildDirectoryState(files) {
  // 从文件路径中自动推导 nifi-data / real_nifi_data 根目录
  if (!state.rootDir) {
    for (const f of files) {
      const path = norm(f.storagePath);
      const idx = path.indexOf("/nifi-data/");
      const idx2 = path.indexOf("/real_nifi_data/");
      if (idx >= 0) { state.rootDir = path.slice(0, idx + "/nifi-data".length); break; }
      if (idx2 >= 0) { state.rootDir = path.slice(0, idx2 + "/real_nifi_data".length); break; }
    }
  }
  // 然后用 KNOWN_NIFI_SUBDIRS 拼出所有子目录
}
```

#### Step 1：顶部下拉菜单

```javascript
// 在 internal.js 初始化时加载用户列表
async function loadUserDropdown() {
    const resp = await fetch('/api/v1/internal/all-users', { credentials: 'include' });
    const data = await resp.json();
    const users = data.data?.users || [];
    const select = document.getElementById('userSelect');
    select.innerHTML = '<option value="">请选择用户</option>';
    users.forEach(u => {
        const label = `${u.username} (${u.deployment_mode === 'private' ? '私有化' : '公有化'})`;
        select.innerHTML += `<option value="${u.username}" data-mode="${u.deployment_mode}">${label}</option>`;
    });
}

// 切换用户事件
document.getElementById('userSelect').addEventListener('change', async (e) => {
    const username = e.target.value;
    if (!username) return;
    const mode = e.target.selectedOptions[0].dataset.mode;
    // 显示/隐藏拉取按钮
    document.getElementById('pullBtn').style.display = mode === 'private' ? 'inline-block' : 'none';
    // 刷新文件树
    await loadAdminFileTree(username);
});
```

#### Step 2：拉取按钮

```javascript
document.getElementById('pullBtn').addEventListener('click', async () => {
    const username = document.getElementById('userSelect').value;
    if (!username) return;
    const resp = await fetch(`/api/v1/internal/private-users/${username}/pull`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({}),
        credentials: 'include'
    });
    const data = await resp.json();
    if (data.code === 0) {
        setStatus('拉取成功', 'success');
        await loadAdminFileTree(username);
    }
});
```
